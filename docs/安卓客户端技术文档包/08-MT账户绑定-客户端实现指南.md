## 一、功能概述

为用户提供绑定 MT4/MT5 交易账户的功能。核心流程：

```
账户列表页 → 点击「绑定」→ 三步向导 → 调用 AddAccount RPC → 返回列表
```

- **绑定时**：后端会实时连接 MT 服务器验证凭据，成功后返回账户信息。
- **绑定后**：用户可在账户列表查看余额/净值/持仓，通过 `GetAccountInfo` 刷新。
- **密码安全**：MT 交易密码以明文传输和存储（连接 MT 服务器必须使用原文）。客户端通过 HTTPS 保护传输安全，本地不缓存密码。

## 二、涉及的 RPC

| Service | RPC | 用途 |
|---|---|---|
| `MT4Service` | `AddAccount` | 绑定 MT4 账户 |
| `MT4Service` | `RemoveAccount` | 删除绑定的 MT4 账户 |
| `MT4Service` | `GetAccountInfo` | 查询 MT4 账户余额/净值/保证金 |
| `MT4Service` | `GetPositions` | 查询 MT4 当前持仓 |
| `MT4Service` | `GetHistory` | 查询 MT4 历史订单 |
| `MT5Service` | `AddAccount` | 绑定 MT5 账户 |
| `MT5Service` | `RemoveAccount` | 删除绑定的 MT5 账户 |
| `MT5Service` | `GetAccountInfo` | 查询 MT5 账户余额/净值/保证金 |
| `MT5Service` | `GetPositions` | 查询 MT5 当前持仓 |
| `MT5Service` | `GetHistory` | 查询 MT5 历史订单 |

**本次实现范围**：`AddAccount` + `RemoveAccount` + `GetAccountInfo`。`GetPositions` 和 `GetHistory` 属于后续迭代。

## 四、RPC 调用封装

在 `data/mt/` 目录创建：

### `data/mt/MtRpcClient.kt`

```kotlin
package com.antclaw.alfq.data.mt

import antclaw.v1.Mt4
import antclaw.v1.Mt5
import com.antclaw.alfq.data.rpc.ConnectTransportProvider
import com.connectrpc.MethodSpec
import com.connectrpc.ResponseMessage
import com.connectrpc.StreamType
import com.connectrpc.getOrThrow

object MtRpcClient {

    private fun client() = ConnectTransportProvider.createProtocolClient()

    // ========== MT4 ==========

    suspend fun addMt4Account(req: AddMt4Request): Mt4Account {
        val proto = Mt4.AddMT4AccountRequest.newBuilder()
            .setServer(req.server)
            .setAccount(req.account)
            .setInvestorPassword(req.investorPassword)
            .setLabel(req.label)
            .setIsDemo(req.isDemo)
            .build()
        val spec = MethodSpec(
            path = "antclaw.v1.MT4Service/AddAccount",
            requestClass = Mt4.AddMT4AccountRequest::class,
            responseClass = Mt4.MT4Account::class,
            streamType = StreamType.UNARY,
        )
        val res: ResponseMessage<Mt4.MT4Account> = client().unary(proto, emptyMap(), spec)
        return res.getOrThrow().toDomain()
    }

    suspend fun removeMt4Account(id: String): Boolean {
        val proto = Mt4.RemoveMT4AccountRequest.newBuilder().setId(id).build()
        val spec = MethodSpec(
            path = "antclaw.v1.MT4Service/RemoveAccount",
            requestClass = Mt4.RemoveMT4AccountRequest::class,
            responseClass = Mt4.RemoveMT4AccountResponse::class,
            streamType = StreamType.UNARY,
        )
        val res: ResponseMessage<Mt4.RemoveMT4AccountResponse> = client().unary(proto, emptyMap(), spec)
        return res.getOrThrow().success
    }

    suspend fun getMt4AccountInfo(id: String): Mt4AccountInfo {
        val proto = Mt4.GetMT4AccountInfoRequest.newBuilder().setId(id).build()
        val spec = MethodSpec(
            path = "antclaw.v1.MT4Service/GetAccountInfo",
            requestClass = Mt4.GetMT4AccountInfoRequest::class,
            responseClass = Mt4.MT4AccountInfo::class,
            streamType = StreamType.UNARY,
        )
        val res: ResponseMessage<Mt4.MT4AccountInfo> = client().unary(proto, emptyMap(), spec)
        return res.getOrThrow().toDomain()
    }

    // ========== MT5 ==========

    suspend fun addMt5Account(req: AddMt5Request): Mt5Account {
        val proto = Mt5.AddMT5AccountRequest.newBuilder()
            .setServer(req.server)
            .setAccount(req.account)
            .setInvestorPassword(req.investorPassword)
            .setLabel(req.label)
            .setIsDemo(req.isDemo)
            .build()
        val spec = MethodSpec(
            path = "antclaw.v1.MT5Service/AddAccount",
            requestClass = Mt5.AddMT5AccountRequest::class,
            responseClass = Mt5.MT5Account::class,
            streamType = StreamType.UNARY,
        )
        val res: ResponseMessage<Mt5.MT5Account> = client().unary(proto, emptyMap(), spec)
        return res.getOrThrow().toDomain()
    }

    suspend fun removeMt5Account(id: String): Boolean {
        val proto = Mt5.RemoveMT5AccountRequest.newBuilder().setId(id).build()
        val spec = MethodSpec(
            path = "antclaw.v1.MT5Service/RemoveAccount",
            requestClass = Mt5.RemoveMT5AccountRequest::class,
            responseClass = Mt5.RemoveMT5AccountResponse::class,
            streamType = StreamType.UNARY,
        )
        val res: ResponseMessage<Mt5.RemoveMT5AccountResponse> = client().unary(proto, emptyMap(), spec)
        return res.getOrThrow().success
    }

    suspend fun getMt5AccountInfo(id: String): Mt5AccountInfo {
        val proto = Mt5.GetMT5AccountInfoRequest.newBuilder().setId(id).build()
        val spec = MethodSpec(
            path = "antclaw.v1.MT5Service/GetAccountInfo",
            requestClass = Mt5.GetMT5AccountInfoRequest::class,
            responseClass = Mt5.MT5AccountInfo::class,
            streamType = StreamType.UNARY,
        )
        val res: ResponseMessage<Mt5.MT5AccountInfo> = client().unary(proto, emptyMap(), spec)
        return res.getOrThrow().toDomain()
    }
}

// Proto → Domain 转换扩展
private fun Mt4.MT4Account.toDomain() = Mt4Account(
    id = id, server = server, account = account, label = label,
    isDemo = isDemo, connected = connected, createdAt = createdAt,

private fun Mt4.MT4AccountInfo.toDomain() = Mt4AccountInfo(

private fun Mt5.MT5Account.toDomain() = Mt5Account(
    id = id, server = server, account = account, label = label,

private fun Mt5.MT5AccountInfo.toDomain() = Mt5AccountInfo(
```

创建 `ui/mt/MtAccountsViewModel.kt`：

```kotlin
package com.antclaw.alfq.ui.mt

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.antclaw.alfq.data.mt.*
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

// 统一的 MT 账户条目（列表展示用）
data class MtAccountItem(
    val id: String,
    val server: String,
    val account: String,
    val label: String,
    val type: String,       // "MT4" | "MT5"
    val isDemo: Boolean,
    val connected: Boolean,
    val balance: Double,
    val equity: Double,
    val createdAt: Long,
)

// UI 状态
data class MtAccountsUiState(
    val accounts: List<MtAccountItem> = emptyList(),
    val loading: Boolean = false,
    val error: String? = null,
    val binding: Boolean = false,     // 正在绑定中
    val bindError: String? = null,    // 绑定失败的错误
    val bindSuccess: Boolean = false, // 绑定成功后短暂置 true
)

class MtAccountsViewModel : ViewModel() {

    private val _state = MutableStateFlow(MtAccountsUiState())
    val state: StateFlow<MtAccountsUiState> = _state

    fun addMt4Account(req: AddMt4Request) {
        viewModelScope.launch {
            _state.update { it.copy(binding = true, bindError = null) }
            try {
                MtRpcClient.addMt4Account(req)
                _state.update { it.copy(binding = false, bindSuccess = true) }
            } catch (e: Exception) {
                _state.update { it.copy(binding = false, bindError = e.message) }
            }
        }
    }

    fun addMt5Account(req: AddMt5Request) {
        viewModelScope.launch {
            _state.update { it.copy(binding = true, bindError = null) }
            try {
                MtRpcClient.addMt5Account(req)
                _state.update { it.copy(binding = false, bindSuccess = true) }
            } catch (e: Exception) {
                _state.update { it.copy(binding = false, bindError = e.message) }
            }
        }
    }

    fun clearBindResult() {
        _state.update { it.copy(bindSuccess = false, bindError = null) }
    }

    fun removeMt4Account(id: String) {
        viewModelScope.launch {
            try {
                MtRpcClient.removeMt4Account(id)
                refreshAccounts()
            } catch (_: Exception) { }
        }
    }

    fun removeMt5Account(id: String) {
        viewModelScope.launch {
            try {
                MtRpcClient.removeMt5Account(id)
                refreshAccounts()
            } catch (_: Exception) { }
        }
    }
```

## 六、UI 实现

### 6.1 三步绑定向导 (`ui/mt/BindMtAccountScreen.kt`)

这是核心页面，实现参考 anttrader 的 `BindAccount.tsx`（React → Compose 映射）。

**页面结构**：

```
Scaffold
├── TopAppBar（标题 + 返回按钮）
└── Column
    ├── StepIndicator（三个圆点 + 连接线）
    └── Card
        ├── Step 1: 选择平台 + 服务器搜索
        │   ├── 平台选择（MT4 / MT5 双按钮）
        │   ├── 服务器搜索输入 + 搜索按钮
        │   └── 搜索结果下拉列表
        ├── Step 2: 输入凭据
        │   ├── 已选服务器信息卡片
        │   ├── 交易账号输入框
        │   └── 密码输入框（只读密码）
        └── Step 3: 确认绑定
            ├── 信息预览卡片
            └── 「确认绑定」按钮
```

**状态变量**：

```kotlin
var step by remember { mutableIntStateOf(1) }            // 1/2/3
var mtType by remember { mutableStateOf("MT4") }          // "MT4" | "MT5"
var server by remember { mutableStateOf("") }             // 搜索关键词
var selectedServer by remember { mutableStateOf("") }     // 选中的服务器名
var account by remember { mutableStateOf("") }            // 交易账号
var investorPassword by remember { mutableStateOf("") }   // 只读密码
var label by remember { mutableStateOf("") }              // 标签（可选）
var isDemo by remember { mutableStateOf(true) }           // 模拟/实盘
```

**Step 1 关键交互**：

1. 点击「搜索」→ 暂用本地模拟的服务器列表（`listOf("ICMarkets-Demo", "ICMarkets-Live", "RoboForex-Demo", ...)`），后续对接 `SearchBroker` RPC。
2. 选择服务器后，「下一步」按钮可用。
3. 选择 MT4/MT5 切换平台类型。

**Step 2 关键交互**：

1. 输入交易账号（纯数字）。
2. 输入只读密码（`investor_password`，**不是主密码**）。
3. 密码框使用 `visualTransformation = PasswordVisualTransformation()`。
4. 两个输入框都非空时，「下一步」按钮可用。

**Step 3 关键交互**：

1. 预览：平台、服务器、账号、标签、模拟/实盘。
2. 点击「确认绑定」→ 根据 `mtType` 调用 `vm.addMt4Account()` 或 `vm.addMt5Account()`。
3. 成功 → `LaunchedEffect(bindSuccess)` 触发 `navController.popBackStack()` 返回。
4. 失败 → 显示错误 Dialog。

**错误处理**：

```kotlin
// 监听 bindError
LaunchedEffect(state.bindError) {
    state.bindError?.let { error ->
        // 显示 AlertDialog 或 Snackbar
    }
}
```

常见错误码参考 `04-客户端开发指导.md` 第三节。

## 十、注意事项

- **密码**：MT 交易密码明文传输（HTTPS 保护），客户端不缓存密码。
- **只读密码**：AntClaw 后端只接受 `investor_password`（只读密码），不能使用 MT 主密码。
- **服务器列表**：当前阶段使用内置模拟列表，后续对接后端 `SearchBroker` RPC。
- **错误状态**：绑定失败时显示可读错误信息，根据 Connect 错误码区分「密码错误」「服务器不可达」「账户不存在」等。
- **不实现管理端**：管理端的 `AdminAccountService` 不在客户端范围内。
