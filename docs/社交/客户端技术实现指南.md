### 1.1 SocialFeedScreen — 社交帖子信息流

**新建文件**: `ui/social/SocialFeedScreen.kt`

```kotlin
@Composable
fun SocialFeedScreen(
    viewModel: SocialFeedViewModel = hiltViewModel(),
    onPostClick: (String) -> Unit = {},     // 点击帖子进入详情
    onUserClick: (String) -> Unit = {},     // 点击用户进入主页
    onSignalClick: (String) -> Unit = {},   // 点击信号
) {
    val state by viewModel.uiState.collectAsState()

    // Tab 切换: 推荐 | 关注
    var selectedTab by remember { mutableIntStateOf(0) }

    Column(modifier = Modifier.fillMaxSize()) {
        TabRow(selectedTabIndex = selectedTab) {
            Tab(selected = selectedTab == 0, onClick = { selectedTab = 0 }, text = { Text("推荐") })
            Tab(selected = selectedTab == 1, onClick = { selectedTab = 1 }, text = { Text("关注") })
        }
        when {
            state.loading -> CircularProgressIndicator()
            state.error != null -> ErrorView(state.error) { viewModel.refresh() }
            else -> LazyColumn {
                items(state.posts, key = { it.id }) { post ->
                    PostCard(post, onLike, onComment, onShare, onUserClick, onPostClick)
                }
            }
        }
    }
}
```

**新建文件**: `ui/social/SocialFeedViewModel.kt`

```kotlin
@HiltViewModel
class SocialFeedViewModel @Inject constructor(
    private val feedRpc: FeedRpcClient
) : ViewModel() {
    data class UiState(
        val posts: List<PostItem> = emptyList(),
        val loading: Boolean = false,
        val error: String? = null,
    )
    private val _uiState = MutableStateFlow(UiState())
    val uiState: StateFlow<UiState> = _uiState.asStateFlow()

    private var feedType = "public" // or "following"

    init { loadFeed() }

    fun loadFeed() {
        viewModelScope.launch {
            _uiState.value = _uiState.value.copy(loading = true, error = null)
            try {
                val posts = when (feedType) {
                    "public"    -> feedRpc.getFeed(pageSize = 20)
                    "following" -> feedRpc.getFollowingFeed(pageSize = 20)
                    else        -> feedRpc.getFeed(pageSize = 20)
                }
                _uiState.value = UiState(posts = posts.map { it.toPostItem() })
            } catch (e: Exception) {
                _uiState.value = _uiState.value.copy(error = e.message)
            }
        }
    }

    fun switchTab(type: String) { feedType = type; loadFeed() }
    fun refresh() { loadFeed() }
}
```

**新建文件**: `ui/components/PostCard.kt`

```kotlin
@Composable
fun PostCard(
    post: PostItem,
    onLike: (String) -> Unit,
    onComment: (String) -> Unit,
    onShare: (String) -> Unit,
    onUserClick: (String) -> Unit,
    onPostClick: (String) -> Unit,
) {
    Card(
        modifier = Modifier.fillMaxWidth().clickable { onPostClick(post.id) },
        shape = RoundedCornerShape(12.dp),
    ) {
        Column(modifier = Modifier.padding(16.dp)) {
            // 头部：头像 + 用户名 + 时间
            Row(verticalAlignment = Alignment.CenterVertically) {
                // Avatar placeholder
                Box(Modifier.size(40.dp).background(Gray, CircleShape))
                Spacer(Modifier.width(12.dp))
                Column {
                    Text(post.authorName, fontWeight = FontWeight.Bold,
                        modifier = Modifier.clickable { onUserClick(post.authorId) })
                    Text(post.timeAgo, style = MaterialTheme.typography.labelSmall, color = Gray)
                }
                Spacer(Modifier.weight(1f))
                // 更多按钮 → 删除/举报
                IconButton(onClick = { showMenu = true }) { Icon(Icons.Default.MoreHoriz, ...) }
                DropdownMenu(expanded = showMenu, ...) {
                    if (post.isOwn) DropdownMenuItem(text = { Text("删除") }, onClick = { onDelete(post.id) })
                    DropdownMenuItem(text = { Text("举报") }, onClick = { onReport(post.id) })
                }
            }
            Spacer(Modifier.height(12.dp))
            // 正文
            if (post.postType == "signal_card") { SignalPostContent(post) }
            else { Text(post.content) }
            // 图片附件（如有）
            if (post.mediaUrls.isNotEmpty()) {
                // LazyRow 横向图片滑动
            }
            Spacer(Modifier.height(12.dp))
            // 互动栏：赞 | 评论 | 转发 | 收藏
            Row(horizontalArrangement = Arrangement.SpaceEvenly, modifier = Modifier.fillMaxWidth()) {
                ActionButton(icon = Icons.Default.Favorite, count = post.likeCount, onClick = { onLike(post.id) })
                ActionButton(icon = Icons.Default.ChatBubble, count = post.commentCount, onClick = { onComment(post.id) })
                ActionButton(icon = Icons.Default.Share, count = post.shareCount, onClick = { onShare(post.id) })
                ActionButton(icon = Icons.Default.Bookmark, onClick = { onBookmark(post.id) })
            }
        }
    }
}
```

### 1.3 评论嵌套 UI

**新建文件**: `ui/components/CommentSection.kt`

```kotlin
@Composable
fun CommentSection(
    comments: List<CommentItem>,
    onReply: (String) -> Unit,  // parent_id
    onLike: (String) -> Unit,
) {
    LazyColumn {
        items(comments) { comment ->
            CommentItem(
                comment = comment,
                depth = 0,
                onReply = onReply,
                onLike = onLike,
            )
        }
    }
}

@Composable
fun CommentItem(comment: CommentItem, depth: Int, onReply: (String) -> Unit, onLike: (String) -> Unit) {
    Row(modifier = Modifier.padding(start = (depth * 24).dp)) {
        // 头像 + 用户名 + 内容 + 回复按钮
        Column {
            Text(comment.authorName, fontWeight = FontWeight.Bold)
            Text(comment.content)
            Row {
                TextButton(onClick = { onLike(comment.id) }) { Text("赞 ${comment.likeCount}") }
                TextButton(onClick = { onReply(comment.id) }) { Text("回复") }
            }
            // 嵌套子评论
            comment.replies.forEach { child ->
                CommentItem(child, depth + 1, onReply, onLike)
            }
        }
    }
}
```

```kotlin
    override fun onNewToken(token: String) {
        // 将 token 发送给服务端: UpdatePushToken RPC

    }

        // 展示本地通知
        showNotification(msg.data["title"], msg.data["body"])
```
